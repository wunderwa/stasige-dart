class CoreConfig {
  final Directory root;
  final BuildConfig build;
  final bool dev;
  final String timekey;
  final String distDir;
  final String pagesDir;

  CoreConfig({
    required this.root,
    required this.build,
    required this.dev,
    required this.timekey,
    required this.distDir,
    required this.pagesDir,
  });

  String pathInPages([String file = '']) => p.join(pagesDir, file);
  String pathInBuild([String file = '']) => p.join(distDir, file);

/// Аналог getConfig({ mono, dev, siteName }) — без мультисайтовости.
CoreConfig getConfig({required bool dev}) {
  final root = Directory.current;
  final buildFile = File(p.join(root.path, 'build.json'));
  final raw = loadYaml(buildFile.readAsStringSync()) as YamlMap;

  final build = BuildConfig(
    langs: (raw['langs'] as YamlList).cast<String>(),
    meta: (raw['meta'] as YamlMap?)?.cast<String, dynamic>() ?? {},
    links: (raw['links'] as YamlMap?)?.cast<String, dynamic>() ?? {},
  );

  final distDir = p.join(root.path, dev ? 'dev' : 'dist');

  return CoreConfig(
    root: root,
    build: build,
    dev: dev,
    timekey: dev ? 'dev' : DateTime.now().millisecondsSinceEpoch.toRadixString(36),
    distDir: distDir,
    pagesDir: p.join(root.path, 'pages'),
  );
}

/// Аналог findPage из core/utils/findPage.ts
PageConfig? findPage(List<PageConfig> pages, {String? pathBase, String? lang}) {
  if (pages.isEmpty) return null;
  if (pathBase == null && lang == null) return pages.first;

  if (pathBase != null && lang != null) {
    final hit = pages.where((p) => p.pathBase == pathBase && p.lang == lang).firstOrNull;
    if (hit != null) return hit;
  }
  if (pathBase != null) {
    return pages.where((p) => p.pathBase == pathBase).firstOrNull ?? pages.first;
  }
  return pages.first;
}

Map<String, dynamic> _readData(String dir) {
  final d = Directory(dir);
  if (!d.existsSync()) return {};
  final out = <String, dynamic>{};
  for (final f in d.listSync().whereType<File>()) {
    if (!f.path.endsWith('.json')) continue;
    final name = f.uri.pathSegments.last.replaceAll('.json', '');
    out[name] = jsonDecode(f.readAsStringSync());
  }
  return out;
}


Map<String, dynamic> _menusToMap(Map<String, List<MenuMainItem>> menus) {
  Map<String, dynamic> item(MenuMainItem i) => {
    'lang': i.lang,
    'path': i.path,
    'pathBase': i.pathBase,
    'menuShort': i.menuShort,
    'menuLong': i.menuLong,
    'title': i.title,
    'children': i.children.map(item).toList(),
  };
  return menus.map((k, v) => MapEntry(k, v.map(item).toList()));
}

Future<void> genPage({
  required CoreConfig cfg,
  required PageConfig page,
  required Menus menus,
  required Map<String, pug.PugHelper> helpers,
}) async {
  final viewsDir = p.join(cfg.root.path, 'views');
  final layoutFile = p.join(viewsDir, 'layouts', '${page.layout}.pug');
  final viewFile = p.join(viewsDir, 'index.pug');

  final options = pug.PugOptions(helpers: helpers, allowLocalAssignments: true);

  final layoutLocals = <String, Object?>{
    'title': page.title,
    'menuShort': page.menuShort,
    'menuLong': page.menuLong,
    'body': page.body,
    'lang': page.lang,
    'path': page.path,
    'pathBase': page.pathBase,
    'langs': cfg.build.langs,
    'mainMenu': _menusToMap(menus.mainMenu),
    'linksByLang': menus.linksByLang,
    'linksByDir': menus.linksByDir,
    'timekey': cfg.timekey,
    'meta': cfg.build.meta,
    'links': cfg.build.links,
    'data': _readData(p.join(cfg.root.path, 'data')),
  };

  final layoutContent = await pug.renderFile(layoutFile, layoutLocals, options);

  final viewLocals = <String, Object?>{
    ...layoutLocals,
    'content': layoutContent,
    'pageLangs': getPageLangs(
      cfg.build.langs,
      page.lang,
      menus.linksByDir[page.pathBase] ?? {},
    ),
  };

  final html = await pug.renderFile(viewFile, viewLocals, options);

  // Убираем ведущий слэш, иначе p.join сбросит distDir
  final relPath = page.path.startsWith('/') ? page.path.substring(1) : page.path;
  final outFile = File(cfg.pathInBuild(p.join(relPath, 'index.html')));
  await outFile.parent.create(recursive: true);
  await outFile.writeAsString(html);
}

import 'package:markdown/markdown.dart' as md;

class _Attrs {
  final String cleanText;
  final String? classes;
  final String? id;
  _Attrs(this.cleanText, this.classes, this.id);
}

_Attrs? _parse(String text) {
  final m = RegExp(r'\{([^}]+)\}\s*$').firstMatch(text);
  if (m == null) return null;
  final raw = m.group(1)!;
  final classes = <String>[];
  String? id;
  for (final part in raw.split(RegExp(r'\s+'))) {
    if (part.startsWith('.')) {
      classes.add(part.substring(1));
    } else if (part.startsWith('#')) {
      id = part.substring(1);
    }
  }
  if (classes.isEmpty && id == null) return null;
  return _Attrs(
    text.substring(0, m.start),
    classes.isEmpty ? null : classes.join(' '),
    id,
  );
}

void _apply(md.Element e) {
  if (e.children == null || e.children!.isEmpty) return;
  final last = e.children!.last;
  if (last is! md.Text) return;
  final a = _parse(last.text);
  if (a == null) return;
  if (a.classes != null) e.attributes['class'] = a.classes!;
  if (a.id != null) e.attributes['id'] = a.id!;
  e.children![e.children!.length - 1] = md.Text(a.cleanText);
}

void _walk(List<md.Node> nodes) {
  for (final n in nodes) {
    if (n is md.Element) {
      _apply(n);
      if (n.children != null) _walk(n.children!);
    }
  }
}

String renderMarkdown(String src) {
  final doc = md.Document(extensionSet: md.ExtensionSet.gitHubWeb);
  final nodes = doc.parseLines(src.split('\n'));
  _walk(nodes);
  return md.HtmlRenderer().render(nodes);
}

import 'dart:io';
import 'package:yaml/yaml.dart';
import 'markdown_attrs.dart';

const separator = '<!--config-->';

Map<String, dynamic> parseMd(String filePath, Map<String, dynamic> build) {
  final content = File(filePath).readAsStringSync();
  final parts = content.split(separator);
  final yamlPart = parts.first.trim()
      .replaceFirst(RegExp(r'^```yaml'), '')
      .replaceFirst(RegExp(r'```$'), '');
  final bodySrc = parts.length > 1 ? parts.sublist(1).join(separator) : '';

  final front = (loadYaml(yamlPart) as YamlMap?)?.cast<String, dynamic>() ?? {};

  return {
    'layout': 'default',
    ...front,
    'body': renderMarkdown(_applyVars(bodySrc, build)),
  };
}

String _applyVars(String src, Map<String, dynamic> build) {
  final map = <String, String>{};
  void walk(dynamic node, String prefix) {
    if (node is Map) {
      node.forEach((k, v) => walk(v, prefix.isEmpty ? k.toString() : '$prefix.$k'));
    } else if (node is List) {
      for (var i = 0; i < node.length; i++) {
        walk(node[i], '$prefix.$i');
      }
    } else {
      map['\${$prefix}'] = node.toString();
    }
  }
  walk(build, '');
  var out = src;
  map.forEach((k, v) => out = out.replaceAll(k, v));
  return out;
}

import 'pages.dart';

class MenuItem {
  final String lang;
  final String path;
  final String pathBase;
  final String menuShort;
  final String menuLong;
  final String title;
  MenuItem({
    required this.lang,
    required this.path,
    required this.pathBase,
    required this.menuShort,
    required this.menuLong,
    required this.title,
  });
}

class MenuMainItem extends MenuItem {
  final List<MenuMainItem> children;
  MenuMainItem({
    required super.lang,
    required super.path,
    required super.pathBase,
    required super.menuShort,
    required super.menuLong,
    required super.title,
    required this.children,
  });
}

class Menus {
  final Map<String, Map<String, MenuItem>> linksByLang;
  final Map<String, Map<String, MenuItem>> linksByDir;
  final Map<String, List<MenuMainItem>> mainMenu;
  Menus({required this.linksByLang, required this.linksByDir, required this.mainMenu});
}

Map<String, Map<String, MenuItem>> _groupByLang(List<PageConfig> pages, List<String> langs) {
  final acc = <String, Map<String, MenuItem>>{};
  for (final pg in pages) {
    if (!langs.contains(pg.lang)) continue;
    acc.putIfAbsent(pg.lang, () => {});
    acc[pg.lang]![pg.pathBase] = MenuItem(
      lang: pg.lang, path: pg.path, pathBase: pg.pathBase,
      menuShort: pg.menuShort, menuLong: pg.menuLong, title: pg.title,
    );
  }
  return acc;
}

Map<String, Map<String, MenuItem>> _groupByDir(List<PageConfig> pages, List<String> langs) {
  final acc = <String, Map<String, MenuItem>>{};
  for (final pg in pages) {
    if (!langs.contains(pg.lang)) continue;
    acc.putIfAbsent(pg.pathBase, () => {});
    acc[pg.pathBase]![pg.lang] = MenuItem(
      lang: pg.lang, path: pg.path, pathBase: pg.pathBase,
      menuShort: pg.menuShort, menuLong: pg.menuLong, title: pg.title,
    );
  }
  return acc;
}

List<MenuMainItem> _groupChildren(Map<String, MenuItem> items) {
  final mapped = items.values.map((it) {
    final dirList = [it.lang, ...it.pathBase.split('/').where((d) => d.isNotEmpty)];
    return {
      'item': it,
      'id': dirList.join(':'),
      'pid': dirList.sublist(0, dirList.length - 1).join(':'),
    };
  }).toList();

  List<MenuMainItem> children(String pid) {
    return mapped.where((m) => m['pid'] == pid).map((m) {
      final it = m['item'] as MenuItem;
      return MenuMainItem(
        lang: it.lang, path: it.path, pathBase: it.pathBase,
        menuShort: it.menuShort, menuLong: it.menuLong, title: it.title,
        children: children(m['id'] as String),
      );
    }).toList();
  }

  return mapped.where((m) => m['pid'] == '').map((m) {
    final it = m['item'] as MenuItem;
    return MenuMainItem(
      lang: it.lang, path: it.path, pathBase: it.pathBase,
      menuShort: it.menuShort, menuLong: it.menuLong, title: it.title,
      children: children(m['id'] as String),
    );
  }).toList();
}

/// Аналог getMenus из core/utils/menus.ts
Menus getMenus(List<PageConfig> pages, List<String> langs) {
  final byLang = _groupByLang(pages, langs);
  final byDir = _groupByDir(pages, langs);
  final mainMenu = byLang.map((lang, items) => MapEntry(lang, _groupChildren(items)));
  return Menus(linksByLang: byLang, linksByDir: byDir, mainMenu: mainMenu);
}

final pageRegExp = RegExp(r'index\.([a-z]{2})\.md$');

/// Аналог getPageConfigList из core/utils/pageConfigs.ts
List<PageConfig> getPageConfigList(CoreConfig cfg) {
  final mainLang = cfg.build.langs.first;
  final dir = Directory(cfg.pagesDir);
  if (!dir.existsSync()) return [];

  return dir
      .listSync(recursive: true)
      .whereType<File>()
      .where((f) => pageRegExp.hasMatch(f.path))
      .map((f) {
    final lang = pageRegExp.firstMatch(f.path)!.group(1)!;
    final rel = p.relative(f.path, from: cfg.pagesDir);
    final pathBase = p.dirname('/$rel');
    final isMain = lang == mainLang;
    final path = '${isMain ? '' : '/$lang'}$pathBase';
    final parsed = parseMd(f.path, {
      'langs': cfg.build.langs,
      'meta': cfg.build.meta,
      'links': cfg.build.links,
    });

    return PageConfig(
      lang: lang,
      src: f.path,
      path: path,
      pathBase: pathBase,
      title: parsed['title']?.toString() ?? '',
      layout: parsed['layout']?.toString() ?? 'default',
      menuShort: parsed['menuShort']?.toString() ?? '',
      menuLong: parsed['menuLong']?.toString() ?? '',
      body: parsed['body']?.toString() ?? '',
    );
  }).toList();
}

/// Публичный API пакета.
/// Аналог src/wrk.ts (упрощённо): поддерживает `wrk -b` и `-D` (dev).
Future<void> runSite({
  required List<String> args,
  Map<String, pug.PugHelper> helpers = const {},
}) async {
  final parser = ArgParser()
    ..addFlag('build', abbr: 'b', negatable: false)
    ..addFlag('dev', abbr: 'D', negatable: false)
    ..addFlag('help', abbr: 'h', negatable: false);

  final argv = parser.parse(args);
  final cmd = argv.rest.isNotEmpty ? argv.rest.first : '';

  if (argv['help'] as bool || cmd == 'help') {
    print('Usage: stasige wrk -b [-D]');
    return;
  }

  if (cmd == 'wrk') {
    if (!(argv['build'] as bool)) {
      print('Nothing to do. Use -b to build.');
      return;
    }
    final cfg = getConfig(dev: argv['dev'] as bool);
    await taskHtml(cfg, helpers);
    print('Built: ${cfg.distDir}');
    return;
  }

  print('Unknown command: $cmd');
}

import 'package:pug_dart/pug_dart.dart' as pug;

Map<String, pug.PugHelper> get helpers => {
  'assets': (args) {
    final path = args.first?.toString() ?? '';
    return '/a/$path';
  },
  'dateFmt': (args) {
    final raw = args.first?.toString() ?? '';
    if (raw.isEmpty) return '';
    final dt = DateTime.tryParse(raw);
    if (dt == null) return raw;
    return dt.toIso8601String().substring(0, 16).replaceAll('T', ' ');
  },
};
